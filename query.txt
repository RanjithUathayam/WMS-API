---Occupied Calculation + Pallet In & Out,Bay Count---
{
SELECT TotalPalletLoc,OccupiedPalletLoc,FreeLoc,ErrorLocWPallet,ErrorLocWOPallet,(OccupiedPalletLoc*100)/TotalPalletLoc  as sumTotal,TotalLoad,TodayInward,TodayOutward,TIw,TOw,totBayCounts
FROM
    (SELECT COUNT(*) as TotalPalletLoc FROM LocationSpecification WHERE PickDelPoint = 0) AS TotalPalletLoc,
    (SELECT COUNT(*) as OccupiedPalletLoc FROM LocationSpecification WHERE PickDelPoint = 0 AND PalletID != '') AS occPlt,
	(select COUNT(*) as FreeLoc from LocationSpecification where PickDelPoint=0 and Blocked=0 and Error=0 and  PalletID is  NULL ) AS FreeLoc,
	(select COUNT(*) as ErrorLocWPallet from LocationSpecification where Error=1 and PickDelPoint=0 and Blocked=0 and ( PalletID!='' or PalletID is not NULL )) AS ErrorLocWPallet,
	(select COUNT(*) as ErrorLocWOPallet from LocationSpecification where Error=1 and PickDelPoint=0 and Blocked=0 and ( PalletID ='' or PalletID is  NULL )) AS ErrorLocWOPallet,
	(select SUM( PalletWeight) as TotalLoad from LocationSpecification where PickDelPoint=0 and Blocked=0 and Quantity>0 and PalletID!='' and PalletID is not NULL) AS TotalLoad,
	(select COUNT(PalletId) as TodayInward from Tbl_PalletCall where Type='Load'  and CONVERT(date,Createddate) >= convert(date,DATEADD(DAY,0,GETDATE()))) AS TodayInward,
	(select COUNT(PalletId) as TodayOutward from Tbl_PalletCall where Type='Unload'  and CONVERT(date,Createddate) >= convert(date,DATEADD(DAY,0,GETDATE()))) AS TodayOutward,
	(select SUM(GrossWeight) as TIw from Inv_PalletDetails_Transaction where InQuantity>0  and CONVERT(date,UpdatedDate) >= convert(date,DATEADD(DAY,0,GETDATE()))) AS TIw,
	(select SUM(GrossWeight) as TOw from Inv_PalletDetails_Transaction where OutQuantity>0  and CONVERT(date,UpdatedDate) >= convert(date,DATEADD(DAY,0,GETDATE()))) AS TOw,
	(SELECT COUNT(*) AS totBayCounts FROM (SELECT DISTINCT Side, Level, Bay FROM LocationSpecification WHERE Blocked = 0 AND PickDelPoint = 0 ) AS subquery) AS totBayCounts;



--Empty Bay Calculation--
SELECT count(BayCount)
FROM (
    SELECT LS.Side,LS.Level,LS.Bay,COUNT(LS.Deep) AS deep,
        (SELECT COUNT(Bay) FROM [LocationSpecification] WHERE PalletID IS NOT NULL AND PickDelPoint = 0 AND Bay = LS.Bay AND Side = LS.Side AND Level = LS.Level) AS BayCount
    FROM LocationSpecification AS LS WHERE  LS.PickDelPoint = 0  GROUP BY LS.Bay,LS.Side,LS.Level
) AS sub
WHERE (SELECT COUNT(Bay) FROM [LocationSpecification] WHERE PalletID IS NOT NULL AND PickDelPoint = 0 AND Bay = sub.Bay AND Side = sub.Side AND Level = sub.Level)=0;
}

CREATE TABLE api_logs (
  id INT PRIMARY KEY IDENTITY,
  method NVARCHAR(255) NOT NULL,
  url NVARCHAR(255) NOT NULL,
  moduleName NVARCHAR(255) NOT NULL,
  queryParameters NVARCHAR(MAX),
  body NVARCHAR(MAX),
  userID NVARCHAR(255) NOT NULL,
  statusCode INT NOT NULL,
  duration INT NOT NULL,
  createdAt DATETIME NOT NULL DEFAULT GETDATE()
);

CREATE INDEX module_name_index ON api_logs (moduleName);
CREATE INDEX url_index ON api_logs (url);
CREATE INDEX method_index ON api_logs (method);
